    <script>
        /* --- JAVASCRIPT - Logica dell'apprendimento --- */

        // 1. Database delle domande
        // Ogni oggetto contiene la domanda (q), le 3 opzioni (options), e l'indice della risposta corretta (correct)
        const questions = [
            {
                q: "Lanci un dado equo a 6 facce. Qual è la probabilità che esca un numero maggiore di 4?",
                options: ["1/3", "1/2", "1/6"],
                correct: 0 // "1/3" è l'opzione all'indice 0 (numeri 5 e 6, quindi 2/6 = 1/3)
            },
            {
                q: "In un mazzo di 52 carte, qual è la probabilità di estrarre un Asso OPPURE un Re?",
                options: ["4/52", "8/52", "2/52"],
                correct: 1 // "8/52" (4 Assi + 4 Re = 8 carte favorevoli)
            },
            {
                q: "Inferenza Statistica: Se un intervallo di confidenza al 95% per la media della popolazione è [10, 20], cosa significa?",
                options: ["Il 95% dei dati cade tra 10 e 20.", "Siamo sicuri al 95% che la vera media sia tra 10 e 20.", "C'è una probabilità del 95% che la media campionaria sia 15."],
                correct: 1 // L'interpretazione corretta dell'IC
            },
            {
                q: "Se eventi A e B sono indipendenti, come si calcola P(A e B)?",
                options: ["P(A) + P(B)", "P(A) * P(B)", "P(A) / P(B)"],
                correct: 1 // Regola del prodotto per eventi indipendenti
            },
            {
                q: "Lanci due monete. Qual è la probabilità di ottenere esattamente una 'Testa'?",
                options: ["1/4", "1/2", "3/4"],
                correct: 1 // Casi possibili: TT, TC, CT, CC. Casi favorevoli (esattamente una T): TC, CT. (2/4 = 1/2)
            }
        ];

        let currentQuestionIndex = 0;
        let isProcessing = false; // Per evitare click multipli durante le transizioni

        // Elementi del DOM
        const questionBox = document.getElementById("question-box");
        const cards = document.querySelectorAll(".card");
        const feedbackMessage = document.getElementById("feedback-message");
        const quizScreen = document.getElementById("quiz-screen");
        const endScreen = document.getElementById("end-screen");

        // Funzione per caricare la domanda corrente
        function loadQuestion() {
            isProcessing = false;
            feedbackMessage.textContent = "";
            
            // Resetta lo stile delle carte
            cards.forEach(card => {
                card.classList.remove("correct", "wrong");
                card.style.pointerEvents = "auto"; // Riabilita i click
            });

            if (currentQuestionIndex < questions.length) {
                const currentQ = questions[currentQuestionIndex];
                questionBox.textContent = currentQ.q;
                
                // Assegna i testi alle carte
                cards.forEach((card, index) => {
                    card.textContent = currentQ.options[index];
                });
            } else {
                showEndScreen();
            }
        }

        // Funzione chiamata quando l'utente clicca su una carta
        function checkAnswer(selectedIndex) {
            if (isProcessing) return;
            isProcessing = true;

            const correctIndex = questions[currentQuestionIndex].correct;
            const selectedCard = cards[selectedIndex];

            // Disabilita i click sulle altre carte per evitare confusione
            cards.forEach(card => card.style.pointerEvents = "none");

            if (selectedIndex === correctIndex) {
                // Risposta Giusta
                selectedCard.classList.add("correct");
                feedbackMessage.style.color = "#28a745";
                feedbackMessage.textContent = "Ottimo lavoro! Risposta esatta! 🎉";
                
                // Aspetta 2 secondi e passa alla prossima domanda
                setTimeout(() => {
                    currentQuestionIndex++;
                    loadQuestion();
                }, 2000);

            } else {
                // Risposta Sbagliata
                selectedCard.classList.add("wrong");
                feedbackMessage.style.color = "#dc3545";
                feedbackMessage.textContent = "Non è corretto. Riprova i calcoli! 🤔";
                
                // Aspetta 1.5 secondi e permetti di riprovare la stessa domanda
                setTimeout(() => {
                     // Rimuovi solo la classe wrong per far riprovare
                    selectedCard.classList.remove("wrong");
                     // Riabilita i click
                    cards.forEach(card => card.style.pointerEvents = "auto");
                    feedbackMessage.textContent = "";
                    isProcessing = false;
                }, 1500);
            }
        }

        function showEndScreen() {
            quizScreen.style.display = "none";
            endScreen.style.display = "block";
        }

        // Avvia il quiz al caricamento della pagina
        loadQuestion();

    </script>
